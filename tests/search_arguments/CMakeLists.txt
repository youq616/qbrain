cmake_minimum_required(VERSION 3.24)
project(qbrain_search_arguments LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(qbrain_search_arguments_tests test.cpp)
target_include_directories(qbrain_search_arguments_tests PRIVATE ../../include)
if(MSVC)
  target_compile_options(qbrain_search_arguments_tests PRIVATE /utf-8 /W4 /EHsc)
else()
  target_compile_options(qbrain_search_arguments_tests PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME search_arguments COMMAND qbrain_search_arguments_tests)
