cmake_minimum_required(VERSION 3.24)
project(qbrain_http_lifecycle_diagnostics LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Native Windows diagnostic only")
endif()
get_filename_component(QBRAIN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(qbrain_http_lifecycle_probe http_lifecycle_probe.cpp)
target_compile_features(qbrain_http_lifecycle_probe PRIVATE cxx_std_20)
target_compile_definitions(qbrain_http_lifecycle_probe PRIVATE QBRAIN_HTTP_DIAGNOSTICS=1 NOMINMAX WIN32_LEAN_AND_MEAN)
target_include_directories(qbrain_http_lifecycle_probe PRIVATE "${QBRAIN_ROOT}/include" "${QBRAIN_ROOT}/third_party")
target_link_libraries(qbrain_http_lifecycle_probe PRIVATE winhttp)
if(MSVC)
  target_compile_options(qbrain_http_lifecycle_probe PRIVATE /utf-8 /EHsc)
  target_link_options(qbrain_http_lifecycle_probe PRIVATE /MANIFEST:NO)
endif()
