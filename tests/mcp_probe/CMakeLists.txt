cmake_minimum_required(VERSION 3.24)
project(qbrain_isolated_probe_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /W3)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(Threads REQUIRED)
add_library(probe_support STATIC "${ROOT}/src/qbrain/util/hash.cpp" "${ROOT}/src/qbrain/util/paths.cpp")
target_include_directories(probe_support PUBLIC "${ROOT}/include" "${ROOT}/third_party")
target_link_libraries(probe_support PUBLIC Threads::Threads)
add_executable(qbrain_mcp_probe_peer "${ROOT}/tests/mcp_probe_peer.cpp")
target_link_libraries(qbrain_mcp_probe_peer PRIVATE probe_support)
add_executable(qbrain_mcp_probe_tests "${ROOT}/tests/test_mcp_probe.cpp")
target_link_libraries(qbrain_mcp_probe_tests PRIVATE probe_support)
enable_testing()
add_test(NAME isolated_mcp_protocol_and_ownership COMMAND qbrain_mcp_probe_tests $<TARGET_FILE:qbrain_mcp_probe_peer>)
