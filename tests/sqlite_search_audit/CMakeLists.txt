cmake_minimum_required(VERSION 3.24)
project(qbrain_sqlite_search_audit LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /W3)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(SQLITE "${ROOT}/third_party/sqlite/sqlite-amalgamation-3460100")
add_library(audit_sqlite STATIC "${SQLITE}/sqlite3.c")
target_include_directories(audit_sqlite PUBLIC "${SQLITE}" "${ROOT}/include" "${ROOT}/third_party")
target_compile_definitions(audit_sqlite PUBLIC SQLITE_ENABLE_FTS5 SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
if(UNIX)
  target_link_libraries(audit_sqlite PUBLIC pthread dl)
endif()
add_executable(qbrain_search_audit_tests "${ROOT}/tests/test_sqlite_search_audit.cpp" "${ROOT}/src/qbrain/util/hash.cpp")
target_link_libraries(qbrain_search_audit_tests PRIVATE audit_sqlite)
enable_testing()
add_test(NAME sqlite_search_audit_contract COMMAND qbrain_search_audit_tests)
