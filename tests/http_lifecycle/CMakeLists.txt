cmake_minimum_required(VERSION 3.24)
project(qbrain_http_lifecycle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(QBRAIN_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /W3 /EHsc)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
foreach(variant IN ITEMS current legacy)
  add_executable(qbrain_http_lifecycle_${variant}
    "${QBRAIN_ROOT}/tests/http_lifecycle_probe.cpp"
    "${QBRAIN_ROOT}/src/qbrain/ai/http_client.cpp")
  target_compile_definitions(qbrain_http_lifecycle_${variant} PRIVATE QBRAIN_HTTP_DIAGNOSTICS=1)
  target_include_directories(qbrain_http_lifecycle_${variant} PRIVATE "${QBRAIN_ROOT}/include" "${QBRAIN_ROOT}/third_party")
  if(variant STREQUAL "legacy")
    target_compile_definitions(qbrain_http_lifecycle_${variant} PRIVATE QBRAIN_HTTP_TEST_EARLY_PARENTS=1)
  endif()
  if(WIN32)
    target_link_libraries(qbrain_http_lifecycle_${variant} PRIVATE winhttp)
  endif()
endforeach()
