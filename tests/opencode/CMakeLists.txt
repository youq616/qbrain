cmake_minimum_required(VERSION 3.24)
project(qbrain_opencode_lifecycle LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /W3)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
find_package(Threads REQUIRED)
add_library(opencode_support STATIC "${ROOT}/src/qbrain/util/hash.cpp" "${ROOT}/src/qbrain/util/paths.cpp")
target_include_directories(opencode_support PUBLIC "${ROOT}/include" "${ROOT}/third_party")
target_link_libraries(opencode_support PUBLIC Threads::Threads)
enable_testing()
foreach(name config audit reconcile write_races)
  add_executable(opencode_${name}_tests "${ROOT}/tests/test_opencode_${name}.cpp")
  target_link_libraries(opencode_${name}_tests PRIVATE opencode_support)
  add_test(NAME opencode_${name} COMMAND opencode_${name}_tests)
endforeach()
