cmake_minimum_required(VERSION 3.24)
project(qbrain_sqlite_check LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /W3)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
set(SQLITE "${ROOT}/third_party/sqlite/sqlite-amalgamation-3460100")
add_library(check_sqlite STATIC "${SQLITE}/sqlite3.c")
target_include_directories(check_sqlite PUBLIC "${SQLITE}" "${ROOT}/include" "${ROOT}/third_party")
target_compile_definitions(check_sqlite PUBLIC SQLITE_ENABLE_FTS5 SQLITE_THREADSAFE=1 SQLITE_OMIT_LOAD_EXTENSION)
if(UNIX)
  target_link_libraries(check_sqlite PUBLIC pthread dl)
endif()
add_executable(qbrain_database_tests "${ROOT}/tests/test_sqlite_check.cpp"
  "${ROOT}/src/qbrain/storage/database.cpp" "${ROOT}/src/qbrain/storage/migrate.cpp"
  "${ROOT}/src/qbrain/util/string_util.cpp" "${ROOT}/src/qbrain/util/hash.cpp")
target_link_libraries(qbrain_database_tests PRIVATE check_sqlite)
enable_testing()
add_test(NAME sqlite_check_contract COMMAND qbrain_database_tests)
