cmake_minimum_required(VERSION 3.24)
project(qbrain_n42_foundation LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
get_filename_component(QBRAIN_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
add_executable(qbrain_n42_foundation test_foundation.cpp "${QBRAIN_ROOT}/src/qbrain/search/rrf.cpp")
target_include_directories(qbrain_n42_foundation PRIVATE "${QBRAIN_ROOT}/include")
target_compile_definitions(qbrain_n42_foundation PRIVATE QBRAIN_N42_STANDALONE=1)
if(MSVC)
  target_compile_options(qbrain_n42_foundation PRIVATE /W4 /utf-8 /EHsc)
  target_link_options(qbrain_n42_foundation PRIVATE /MANIFEST:NO)
else()
  target_compile_options(qbrain_n42_foundation PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME n42_foundation COMMAND qbrain_n42_foundation)

option(QBRAIN_N42_SQLITE_ADAPTER_TEST "Test production search code using a test-only SQLite adapter" OFF)
if(QBRAIN_N42_SQLITE_ADAPTER_TEST)
  set(SQLITE_SRC "${QBRAIN_ROOT}/third_party/sqlite/sqlite-amalgamation-3460100/sqlite3.c")
  if(EXISTS "${SQLITE_SRC}")
    enable_language(C)
    add_library(qbrain_n42_sqlite STATIC "${SQLITE_SRC}")
    get_filename_component(SQLITE_INC "${SQLITE_SRC}" DIRECTORY)
    target_include_directories(qbrain_n42_sqlite PUBLIC "${SQLITE_INC}")
    target_compile_definitions(qbrain_n42_sqlite PRIVATE SQLITE_ENABLE_FTS5 SQLITE_OMIT_LOAD_EXTENSION)
    set(N42_SQLITE_TARGET qbrain_n42_sqlite)
  else()
    find_package(SQLite3 REQUIRED)
    set(N42_SQLITE_TARGET SQLite::SQLite3)
  endif()
  add_executable(qbrain_n42_search_adapter test_search_adapter.cpp
    "${QBRAIN_ROOT}/src/qbrain/search/hybrid.cpp"
    "${QBRAIN_ROOT}/src/qbrain/search/rrf.cpp"
    "${QBRAIN_ROOT}/src/qbrain/search/vector.cpp")
  target_include_directories(qbrain_n42_search_adapter BEFORE PRIVATE
    "${CMAKE_CURRENT_LIST_DIR}/support" "${QBRAIN_ROOT}/include")
  target_link_libraries(qbrain_n42_search_adapter PRIVATE ${N42_SQLITE_TARGET})
  if(MSVC)
    target_compile_options(qbrain_n42_search_adapter PRIVATE /utf-8 /EHsc)
    target_link_options(qbrain_n42_search_adapter PRIVATE /MANIFEST:NO)
  endif()
  add_test(NAME n42_search_sqlite_test_adapter COMMAND qbrain_n42_search_adapter)
endif()
