cmake_minimum_required(VERSION 3.24)
project(qbrain_token_cost LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /W3)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(cost_support STATIC "${ROOT}/src/qbrain/util/hash.cpp")
target_include_directories(cost_support PUBLIC "${ROOT}/include" "${ROOT}/third_party")
add_executable(qbrain_cost_tests "${ROOT}/tests/test_token_cost.cpp")
target_link_libraries(qbrain_cost_tests PRIVATE cost_support)
enable_testing()
add_test(NAME token_cost_arithmetic_and_contract COMMAND qbrain_cost_tests)
