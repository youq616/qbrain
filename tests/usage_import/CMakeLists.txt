cmake_minimum_required(VERSION 3.24)
project(qbrain_usage_import LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
if(MSVC)
  add_compile_options(/utf-8 /EHsc /W3)
  add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
add_library(usage_import_support STATIC "${ROOT}/src/qbrain/util/hash.cpp")
target_include_directories(usage_import_support PUBLIC "${ROOT}/include" "${ROOT}/third_party")
add_executable(qbrain_usage_import_tests "${ROOT}/tests/test_provider_usage.cpp")
target_link_libraries(qbrain_usage_import_tests PRIVATE usage_import_support)
enable_testing()
add_test(NAME provider_usage_contract COMMAND qbrain_usage_import_tests)
